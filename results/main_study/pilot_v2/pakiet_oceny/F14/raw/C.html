def put():
  return  list(map(int, input().split()))


def dfs(x,flag=1):
    s,vis,ans   = [x],[0]*n,['+']*m
    vis[x]= 1
    while s:
        i = s.pop()
        for  j,k in  graph[i]:
            if  vis[j]==0:
                if k*flag < 0:
                    ans[abs(k)-1]='-'
                elif flag*k > 0:
                    ans[abs(k) - 1] = '+'
                if 1==flag or 0==k:
                    s.append(j)
                    vis[j]=1
    return  ''.join(ans), sum(vis)

n,m,s=put()
graph, k = [[] for i in range(n)], 1
for _ in range(0, m):
    z,x,y = put()
    x,y = x - 1,y - 1
    if not z==1:
        graph[x].append((y, k))
        graph[y].append((x,-k))
        k+=1
    else:
        graph[x].append((y, 0))
m=k-1
x,y=dfs(s-1, 1)
print(y)
print(x)
